- [ ] **Step 2 : Créer `apps/web/src/lib/source.ts`**

Fumadocs `loader()` scans MDX files. Check the installed version of fumadocs-core to confirm the correct import path. For fumadocs-core v14+:

```typescript
// apps/web/src/lib/source.ts
import { loader } from "fumadocs-core/source";
import { createMDXSource } from "fumadocs-mdx";
import type { InferMetaType, InferPageType } from "fumadocs-core/source";

export const docsSource = loader({
  baseUrl: "/docs",
  source: createMDXSource(
    // @ts-expect-error -- generated by fumadocs-mdx at build time
    // eslint-disable-next-line @typescript-eslint/no-require-imports
    require("../../.source/docs").default,
  ),
});

export const blogSource = loader({
  baseUrl: "/blog",
  source: createMDXSource(
    // @ts-expect-error -- generated by fumadocs-mdx at build time
    // eslint-disable-next-line @typescript-eslint/no-require-imports
    require("../../.source/blog").default,
  ),
});

export type DocsPage = InferPageType<typeof docsSource>;
export type BlogPage = InferPageType<typeof blogSource>;
export type DocsMeta = InferMetaType<typeof docsSource>;
````

Note: The `.source/` directory is auto-generated by `fumadocs-mdx` during the build/dev process. At type-check time it may not exist yet — that's normal and is why we use `@ts-expect-error`.

```tsx
import { notFound } from "next/navigation";
import { Nav } from "../../../components/nav";
import { Footer } from "../../../components/footer";

// Static map — mirrors content/blog/ directory
const slugMap: Record<
  string,
  () => Promise<{
    default: React.ComponentType;
    frontmatter: Record<string, unknown>;
  }>
> = {
  "getting-started": () =>
    import("../../../../content/blog/getting-started.mdx"),
};

export function generateStaticParams() {
  return Object.keys(slugMap).map((slug) => ({ slug }));
}

export default async function BlogPostPage({
  params,
}: {
  params: Promise<{ slug: string }>;
}) {
  const { slug } = await params;
  const importer = slugMap[slug];
  if (!importer) notFound();

  const { default: MDXContent, frontmatter } = await importer();
  const fm = frontmatter as { title?: string; date?: string };

  return (
    <>
      <Nav />
      <main className="mx-auto max-w-2xl px-4 py-20">
        <p className="text-sm text-muted-foreground mb-2">{fm.date}</p>
        <h1 className="text-4xl font-bold mb-8">{fm.title}</h1>
        <div className="prose prose-neutral dark:prose-invert max-w-none">
          <MDXContent />
        </div>
      </main>
      <Footer />
    </>
  );
}

import { notFound } from "next/navigation";
import {
  DocsPage,
  DocsBody,
  DocsTitle,
  DocsDescription,
} from "fumadocs-ui/page";
import { docsSource } from "../../../lib/source";

export default async function Page({
  params,
}: {
  params: Promise<{ slug?: string[] }>;
}) {
  const { slug } = await params;
  const page = docsSource.getPage(slug);
  if (!page) notFound();

  const MDX = page.data.body;

  return (
    <DocsPage toc={page.data.toc}>
      <DocsTitle>{page.data.title}</DocsTitle>
      <DocsDescription>{page.data.description}</DocsDescription>
      <DocsBody>
        <MDX />
      </DocsBody>
    </DocsPage>
  );
}

export function generateStaticParams() {
  return docsSource.generateParams();
}
```

Note: `DocsPage`, `DocsBody`, `DocsTitle`, `DocsDescription` are fumadocs-ui components. If the import path `fumadocs-ui/page` doesn't exist, check `node_modules/fumadocs-ui/dist/` for the correct path.
